cmake_minimum_required(VERSION 3.24)
project(SporeMPAudioCapture LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Original SPORE process audio capture requires Windows.")
endif()
add_executable(SporeMP.AudioCapture ../capture-audio.cpp)
target_compile_features(SporeMP.AudioCapture PRIVATE cxx_std_17)
target_compile_definitions(SporeMP.AudioCapture PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_options(SporeMP.AudioCapture PRIVATE /W4 /permissive- /EHsc)
target_link_libraries(SporeMP.AudioCapture PRIVATE mmdevapi ole32)
