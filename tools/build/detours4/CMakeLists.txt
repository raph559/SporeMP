cmake_minimum_required(VERSION 3.24)
project(SporeMPDetours VERSION 4.0.1 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Detours requires the pinned MSVC Win32 toolchain.")
endif()
set(CMAKE_CONFIGURATION_TYPES "Debug;Release" CACHE STRING "Supported configurations" FORCE)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
file(READ "${REPO_ROOT}/config/dependencies.lock.json" DEPENDENCIES)
string(JSON REVISION GET "${DEPENDENCIES}" native_loader commit)
set(LOADER_ROOT "${REPO_ROOT}/external/ModAPI-Launcher-Kit")
execute_process(COMMAND git -C "${LOADER_ROOT}" rev-parse HEAD OUTPUT_VARIABLE ACTUAL
  OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT ACTUAL STREQUAL REVISION)
  message(FATAL_ERROR "Detours source revision differs from the dependency lock.")
endif()
set(DETOURS_ROOT "${LOADER_ROOT}/ModAPI.DLLInjector/Detours/src")
add_library(sporemp_detours STATIC "${DETOURS_ROOT}/detours.cpp" "${DETOURS_ROOT}/modules.cpp"
  "${DETOURS_ROOT}/disasm.cpp" "${DETOURS_ROOT}/image.cpp" "${DETOURS_ROOT}/creatwth.cpp")
target_include_directories(sporemp_detours PUBLIC "${DETOURS_ROOT}")
target_compile_definitions(sporemp_detours PRIVATE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
