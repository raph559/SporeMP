# Standalone HOST build, also includable from the repository root.
cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(SporeMPNetwork LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  include(CTest)
endif()
add_library(sporemp_network STATIC protocol.cpp session.cpp tls.cpp channel.cpp peer.cpp coordinator.cpp)
target_include_directories(sporemp_network PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(sporemp_network PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0602 SECURITY_WIN32)
target_link_libraries(sporemp_network PUBLIC ws2_32 secur32 crypt32 ncrypt bcrypt)
target_compile_options(sporemp_network PRIVATE /W4 /WX)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp")
  add_executable(SporeMP.Coordinator main.cpp)
  set_target_properties(SporeMP.Coordinator PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
  target_link_libraries(SporeMP.Coordinator PRIVATE sporemp_network)
  target_compile_options(SporeMP.Coordinator PRIVATE /W4 /WX)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/unit/network_tests.cpp")
  add_executable(sporemp_network_tests ../../tests/unit/network_tests.cpp)
  target_link_libraries(sporemp_network_tests PRIVATE sporemp_network)
  target_compile_options(sporemp_network_tests PRIVATE /W4 /WX)
  add_test(NAME network_host COMMAND sporemp_network_tests)
  set_tests_properties(network_host PROPERTIES LABELS "host;fixture;not-native" TIMEOUT 60)
endif()
