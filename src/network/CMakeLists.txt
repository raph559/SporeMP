# Standalone HOST build, also includable from the repository root.
cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(SporeMPNetwork LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  include(CTest)
endif()
add_executable(SporeMP.ContentClient content_client_main.cpp)
set_target_properties(SporeMP.ContentClient PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
target_link_libraries(SporeMP.ContentClient PRIVATE sporemp_network)
target_compile_options(SporeMP.ContentClient PRIVATE /W4 /WX)
add_library(sporemp_network STATIC protocol.cpp session.cpp tls.cpp channel.cpp peer.cpp coordinator.cpp world_identity.cpp content_registry.cpp content_wire.cpp content_control.cpp durable_store.cpp checkpoint_artifact.cpp location_ledger.cpp transfer_control.cpp)
target_include_directories(sporemp_network PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(sporemp_network PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0602 SECURITY_WIN32)
target_link_libraries(sporemp_network PUBLIC ws2_32 secur32 crypt32 ncrypt bcrypt)
target_compile_options(sporemp_network PRIVATE /W4 /WX)
add_executable(sporemp_durable_store_tests ../../tests/unit/durable_store_tests.cpp)
target_link_libraries(sporemp_durable_store_tests PRIVATE sporemp_network)
target_compile_options(sporemp_durable_store_tests PRIVATE /W4 /WX)
add_test(NAME durable_store_host COMMAND sporemp_durable_store_tests)
set_tests_properties(durable_store_host PROPERTIES LABELS "host;fixture;not-native" TIMEOUT 60)
