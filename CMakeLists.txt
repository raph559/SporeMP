cmake_minimum_required(VERSION 3.24)
project(SporeMP VERSION 0.0.1 LANGUAGES CXX)
set(CMAKE_CONFIGURATION_TYPES "Debug;Release" CACHE STRING "Supported pinned SDK configurations" FORCE)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "M01 bridge requires MSVC Win32. Use tools/build/build.ps1.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(SDK_ROOT "${CMAKE_SOURCE_DIR}/external/Spore-ModAPI")
file(READ "${CMAKE_SOURCE_DIR}/config/dependencies.lock.json" DEPENDENCIES)
string(JSON SDK_REVISION GET "${DEPENDENCIES}" modapi commit)
execute_process(COMMAND git -C "${SDK_ROOT}" rev-parse HEAD OUTPUT_VARIABLE SDK_ACTUAL OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT SDK_ACTUAL STREQUAL SDK_REVISION)
  message(FATAL_ERROR "SDK revision differs from the dependency lock.")
endif()
file(READ "${CMAKE_SOURCE_DIR}/config/compatibility.candidate.json" COMPATIBILITY)
string(JSON GAME_SHA256 GET "${COMPATIBILITY}" executable sha256)
configure_file(src/bridge/build_identity.h.in generated/build_identity.h @ONLY)
add_library(modapi SHARED IMPORTED)
add_library(modapi_base STATIC IMPORTED)
foreach(CONFIG Debug Release)
  string(TOUPPER "${CONFIG}" UPPER_CONFIG)
  set_property(TARGET modapi PROPERTY IMPORTED_IMPLIB_${UPPER_CONFIG} "${CMAKE_SOURCE_DIR}/build/sdk/${CONFIG}/SporeModAPI.lib")
  set_property(TARGET modapi PROPERTY IMPORTED_LOCATION_${UPPER_CONFIG} "${CMAKE_SOURCE_DIR}/build/sdk/${CONFIG}/SporeModAPI.dll")
  set_property(TARGET modapi_base PROPERTY IMPORTED_LOCATION_${UPPER_CONFIG} "${CMAKE_SOURCE_DIR}/build/sdk-base/${CONFIG}/SporeModAPIBase.lib")
endforeach()
add_library(sporemp_diagnostics STATIC src/bridge/diagnostics.cpp)
target_include_directories(sporemp_diagnostics PUBLIC src/bridge "${CMAKE_BINARY_DIR}/generated")
target_link_libraries(sporemp_diagnostics PUBLIC bcrypt)
target_compile_definitions(sporemp_diagnostics PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(sporemp_diagnostics PRIVATE /W4 /WX)
add_library(SporeMP.Bridge SHARED src/bridge/dllmain.cpp)
target_include_directories(SporeMP.Bridge SYSTEM PRIVATE "${SDK_ROOT}/Spore ModAPI" "${SDK_ROOT}/EASTL-3.02.01/include" "${SDK_ROOT}/EASTL-3.02.01/test/packages/EABase/include/Common" "${SDK_ROOT}/EASTL-3.02.01/test/packages/EAAssert/include" "${SDK_ROOT}/EASTL-3.02.01/test/packages/EAStdC/include" "${SDK_ROOT}/Detours/include")
target_link_libraries(SporeMP.Bridge PRIVATE modapi_base modapi sporemp_diagnostics)
target_compile_options(SporeMP.Bridge PRIVATE /W4)
# No post-build install/copy to a game/launcher directory and no automatic launch.
include(CTest)
add_executable(sporemp_diagnostics_tests tests/unit/diagnostics_tests.cpp)
target_link_libraries(sporemp_diagnostics_tests PRIVATE sporemp_diagnostics)
target_compile_options(sporemp_diagnostics_tests PRIVATE /W4 /WX)
add_test(NAME diagnostics_host COMMAND sporemp_diagnostics_tests)
set_tests_properties(diagnostics_host PROPERTIES LABELS "host;not-native")
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/native/$<CONFIG>/native_identity.h"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/build/generate-native-identity.py" "${CMAKE_SOURCE_DIR}" "$<CONFIG>" "${CMAKE_BINARY_DIR}/native/$<CONFIG>/native_identity.h"
  DEPENDS SporeMP.Bridge config/compatibility.candidate.json tools/build/generate-native-identity.py
    "${CMAKE_SOURCE_DIR}/build/injector/$<CONFIG>/ModAPI.DLLInjector.dll"
    "${CMAKE_SOURCE_DIR}/build/sdk/$<CONFIG>/SporeModAPI.dll")
add_executable(SporeMP.NativeHost src/bridge/native_host.cpp "${CMAKE_BINARY_DIR}/native/$<CONFIG>/native_identity.h")
target_include_directories(SporeMP.NativeHost PRIVATE "${CMAKE_BINARY_DIR}/native/$<CONFIG>")
target_link_libraries(SporeMP.NativeHost PRIVATE sporemp_diagnostics shell32 advapi32 userenv)
target_compile_options(SporeMP.NativeHost PRIVATE /W4 /WX)
add_executable(SporeMP.TraceExtract src/bridge/trace_extract.cpp)
target_compile_definitions(SporeMP.TraceExtract PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(SporeMP.TraceExtract PRIVATE advapi32 tdh ole32)
target_compile_options(SporeMP.TraceExtract PRIVATE /W4 /WX)
add_test(NAME tooling_unit COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s tests/unit -p test_*.py -v)
set_tests_properties(tooling_unit PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" LABELS "fixture;not-native")
add_test(NAME launcher_host COMMAND "${CMAKE_SOURCE_DIR}/build/launcher-tests/$<CONFIG>/SporeMP.Launcher.Tests.exe" "${CMAKE_SOURCE_DIR}/build/launcher/$<CONFIG>/launcher.runtime.json")
set_tests_properties(launcher_host PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" LABELS "host;not-native")
